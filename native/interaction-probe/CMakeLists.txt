cmake_minimum_required(VERSION 3.20)
project(interaction_probe VERSION 0.6.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(vendor/minhook-1.3.4 EXCLUDE_FROM_ALL)
add_library(interaction_probe SHARED probe.cpp)
target_include_directories(interaction_probe PRIVATE vendor/shroudtopia)
target_compile_definitions(interaction_probe PRIVATE NOMINMAX)
target_link_libraries(interaction_probe PRIVATE minhook advapi32 user32)
target_compile_options(interaction_probe PRIVATE /W4 /WX /EHsc /guard:cf)
add_executable(probe_tests tests.cpp)
target_compile_options(probe_tests PRIVATE /W4 /WX /EHsc)
add_executable(probe_runtime_tests runtime_tests.cpp)
target_include_directories(probe_runtime_tests PRIVATE vendor/shroudtopia)
target_compile_definitions(probe_runtime_tests PRIVATE NOMINMAX)
target_link_libraries(probe_runtime_tests PRIVATE minhook advapi32 user32)
target_compile_options(probe_runtime_tests PRIVATE /W4 /WX /EHsc /guard:cf)
enable_testing()
add_test(NAME probe_events COMMAND probe_tests)
add_test(NAME probe_runtime COMMAND probe_runtime_tests)
