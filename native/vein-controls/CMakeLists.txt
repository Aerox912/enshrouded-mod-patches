cmake_minimum_required(VERSION 3.20)
project(vein_controls VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(vmkeys SHARED proxy.cpp vmkeys.def)
target_link_libraries(vmkeys PRIVATE user32)
add_executable(vein_controls_tests tests.cpp)
target_link_libraries(vein_controls_tests PRIVATE user32)
foreach(target vmkeys vein_controls_tests)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /guard:cf)
endforeach()
enable_testing()
add_test(NAME vein_controls COMMAND vein_controls_tests)
