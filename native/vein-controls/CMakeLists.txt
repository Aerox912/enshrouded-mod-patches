cmake_minimum_required(VERSION 3.20)
project(vein_controls VERSION 1.2.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SERVER_COMPAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../enshrouded-server-compat" CACHE PATH "Pinned shared vein radius source and MinHook")
add_subdirectory("${SERVER_COMPAT_ROOT}/vendor/minhook-1.3.4" minhook EXCLUDE_FROM_ALL)
add_library(vmkeys SHARED proxy.cpp native_bridge.cpp vmkeys.def "${SERVER_COMPAT_ROOT}/src/vein_radius.cpp")
target_include_directories(vmkeys PRIVATE "${SERVER_COMPAT_ROOT}/src")
target_link_libraries(vmkeys PRIVATE user32 gdi32 bcrypt minhook)
add_executable(vein_controls_tests tests.cpp)
target_link_libraries(vein_controls_tests PRIVATE user32)
add_executable(vein_bridge_tests bridge_tests.cpp)
target_link_libraries(vein_bridge_tests PRIVATE user32)
foreach(target vmkeys vein_controls_tests vein_bridge_tests)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /guard:cf)
endforeach()
set(VEIN_RADIUS_FIXTURE "" CACHE FILEPATH "Private expanded client DLL for actual-callback tests")
if(VEIN_RADIUS_FIXTURE)
  add_test(NAME vein_radius_client_bridge COMMAND vein_bridge_tests "${VEIN_RADIUS_FIXTURE}")
endif()
set(VEIN_RADIUS_ORIGINAL_FIXTURE "" CACHE FILEPATH "Private expanded original-language client DLL")
if(VEIN_RADIUS_ORIGINAL_FIXTURE)
  add_test(NAME vein_radius_original_client_bridge COMMAND vein_bridge_tests "${VEIN_RADIUS_ORIGINAL_FIXTURE}")
endif()
set(VEIN_STOCK_FIXTURE "" CACHE FILEPATH "Private stock-capacity DLL with helper import")
if(VEIN_STOCK_FIXTURE)
  add_test(NAME vein_unsupported_client_rejected COMMAND vein_bridge_tests "${VEIN_STOCK_FIXTURE}" reject)
endif()
enable_testing()
foreach(mode hold toggle always)
  add_test(NAME vein_controls_${mode} COMMAND vein_controls_tests ${mode} "${VEIN_RADIUS_FIXTURE}")
endforeach()
